/* Return the index of the first element of a[0..n) that is >= key, or n
 * when every element is smaller. a is sorted ascending. */
static size_t lower_bound(size_t n, const int a[static n], int key) {
    size_t lo = 0, hi = n;
    while (lo < hi) {
        size_t mid = lo + (hi - lo) / 2;
        if (a[mid] < key)
            lo = mid + 1;
        else
            hi = mid;
    }
    return lo;
}
